#!/usr/bin/env bash
# Usage:
#   ./loadAStocksYears.sh 2020 2021  Load the listed years at startup.
#   ./loadAStocksYears.sh           Load the previous two years and current year.
# After startup, the same command sends the years to the running process:
#   ./loadAStocksYears.sh 2022 2023
set -euo pipefail

source /home/toby/env.sh

readonly root_dir="${VENDOR}/mydata/stock/AStock"
readonly q_bin="${QHOME}/bin/q"
readonly q_loader="${root_dir}/loadAStocksYears.q"
readonly port=5082
cd "$root_dir"
current_year=$(date +%Y)

if (($# == 0)); then
    set -- "$((current_year - 2))" "$((current_year - 1))" "$current_year"
fi

valid_years=()
for year in "$@"; do
    if [[ "$year" =~ ^[0-9]{4}$ ]] && ((10#$year >= 1990 && 10#$year <= current_year)); then
        valid_years+=("$year")
    else
        printf 'Skipping invalid year: %s (allowed range: 1990-%s)\n' "$year" "$current_year" >&2
    fi
done


if [[ ! -x "$q_bin" ]]; then
    printf 'q executable not found or not executable: %s\n' "$q_bin" >&2
    exit 1
fi

if ((${#valid_years[@]})); then
    printf 'Loading years into in-memory md_daily: %s\n' "${valid_years[*]}"
else
    printf 'No valid years supplied; starting the in-memory KDB process without loading data.\n'
fi

# Load into the existing in-memory process when it is already listening.
if { exec 3<>"/dev/tcp/127.0.0.1/${port}"; } 2>/dev/null; then
    exec 3>&-
    exec "$q_bin" "$q_loader" client "${valid_years[@]}"
fi

exec "$q_bin" "$q_loader" "${valid_years[@]}" -p "$port"
