/ Usage:
/   Start with explicit years: loadAStocksYears.sh 2020 2021
/   Start with default recent years: loadAStocksYears.sh
/   After startup, load additional years with: loadAStocksYears.sh 2022 2023
/ Data stays in this process's in-memory md_daily table.
\l loadOneAStock1Year.q

/ The shell script uses this mode to add years to an existing port 5082 process.
if[0<count .z.x;
    if[`client~first .z.x;
        years:1_ .z.x;
        h:hopen `:localhost:5082;
        h (`loadYears; years);
        hclose h;
        exit 0;
    ];
 ];

stockRoot:getenv `MYDATA_STOCK;
stockRoot,:"/AStock";

loadCodes:{[file;prefix]
    c:read0 `$file;
    c:c where not c~\:"";
    `$(prefix,/:-6#/:c)
 };

symbols:raze (loadCodes[stockRoot,"/sh.code";"sh"];
               loadCodes[stockRoot,"/sz.code";"sz"];
               loadCodes[stockRoot,"/bj.code";"bj"]);
years:.z.x;
