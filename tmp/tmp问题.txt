#!/bin/bash
# download-artifacts.sh - 仅下载原始文件，不处理

set -e

ARCH="$(uname -m)"
S3="https://s3.amazonaws.com/spec.ccfc.min"

echo "🔍 检测架构: $ARCH"

# 获取最新 CI 构建目录
CI_ARTIFACTS_PREFIX=$(curl -fsSL "$S3?list-type=2&prefix=firecracker-ci/&delimiter=/" \
    | grep -oP "(?<=<Prefix>)firecracker-ci/[0-9]{8}-[^/]+/(?=</Prefix>)" \
    | sort | tail -1)

echo "📁 使用 CI 前缀: ${CI_ARTIFACTS_PREFIX}"

# 下载内核
latest_kernel_key=$(curl -fsSL "$S3?list-type=2&prefix=${CI_ARTIFACTS_PREFIX}${ARCH}/vmlinux-" \
    | grep -oP "(?<=<Key>)(${CI_ARTIFACTS_PREFIX}${ARCH}/vmlinux-[0-9]+\.[0-9]+\.[0-9]{1,3})(?=</Key>)" \
    | sort -V | tail -1)
echo "📥 下载内核: $(basename "$latest_kernel_key")"
wget -q "$S3/${latest_kernel_key}" -O "$(basename "$latest_kernel_key")"

# 下载 Ubuntu rootfs
latest_ubuntu_key=$(curl -fsSL "$S3?list-type=2&prefix=${CI_ARTIFACTS_PREFIX}${ARCH}/ubuntu-" \
    | grep -oP "(?<=<Key>)(${CI_ARTIFACTS_PREFIX}${ARCH}/ubuntu-[0-9]+\.[0-9]+\.squashfs)(?=</Key>)" \
    | sort -V | tail -1)
echo "📥 下载 rootfs: $(basename "$latest_ubuntu_key")"
wget -q "$S3/$latest_ubuntu_key" -O "$(basename "$latest_ubuntu_key")"

echo
echo "✅ 下载完成！请将以下文件拷贝到无网机器："
ls vmlinux-* ubuntu-*.squashfs
