[Unit]
Description=Snapshot Go Unix Socket Service
After=network.target

[Service]
Type=simple
User=toby
Group=toby

Environment=APP_SOCK=/tmp/snapshot.sock
Environment=APP_PID=/tmp/snapshot.pid

ExecStart=/tmp/snapshot
Restart=on-failure
RestartSec=3

RuntimeDirectory=Snapshot
RuntimeDirectoryMode=0755

TimeoutStopSec=10

StandardOutput=journal
StandardError=journal

LimitNOFILE=65535

[Install]
WantedBy=multi-user.target
