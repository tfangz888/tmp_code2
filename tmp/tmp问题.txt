version: '3.8'

services:
  mariadb:
    image: mariadb
    container_name: mariadb
    restart: unless-stopped
    
    environment:
      MARIADB_ROOT_PASSWORD: root123
      MARIADB_DATABASE: myapp
      MARIADB_USER: appuser
      MARIADB_PASSWORD: pass123
    
    ports:
      - "3306:3306"  # 如需外部访问
    
    volumes:
      - /aaa/bbb:/var/lib/mysql
    
    command: >
      --character-set-server=utf8mb4
      --collation-server=utf8mb4_unicode_ci
      --skip-name-resolve
      --performance-schema=OFF
      --innodb-buffer-pool-size=64M
      --max-connections=20
    
    healthcheck:  # 可选：健康检查
      test: ["CMD", "healthcheck.sh", "--connect"]
      interval: 30s
      timeout: 10s
      retries: 3



# 2. 启动服务
# docker-compose up -d

# 3. 查看状态
# docker-compose ps

# 4. 测试连接
# docker-compose exec mariadb mysql -u root -proot123 -e "SHOW DATABASES;"

# 5. 停止服务
# docker-compose stop

# 6. 重启服务
# docker-compose restart
