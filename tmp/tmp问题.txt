#!/bin/bash
# 检查所有挂载点使用率，超过阈值就记录日志
THRESHOLD=70
# 定义要排除的硬盘设备（用空格分隔）
EXCLUDE_DISKS="tmpfs devtmpfs udev sysfs proc overlay"

df -h | while read filesystem size used avail percent mountpoint; do
    # 跳过标题行和特殊文件系统
    if [[ "$percent" =~ ^[0-9]+%$ ]] && [ "${percent%\%}" -gt $THRESHOLD ]; then
        # 检查是否在排除的硬盘列表中
        skip=false
        for exclude in $EXCLUDE_DISKS; do
            if [[ "$filesystem" == "$exclude"* ]]; then
                skip=true
                break
            fi
        done
        
        # 如果不在排除列表中，则输出
        if [ "$skip" = false ]; then
            echo "$(date): 硬盘 $filesystem 挂载在 $mountpoint 使用率 $percent 超过${THRESHOLD}%"
        fi
    fi
done






sudo nano /etc/systemd/system/disk-check.service

[Unit]
Description=Check Disk Usage
After=multi-user.target

[Service]
Type=oneshot
ExecStart=/usr/local/bin/check-disk.sh

[Install]
WantedBy=multi-user.target
