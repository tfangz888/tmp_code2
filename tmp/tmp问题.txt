# 先要在 tickerplant的符号表里定义表结构
# kdb/tickerplant/tick/symA.q
# trade:([] time:`timestamp$(); sym:`symbol$(); price:`float$(); size:`long$())

import os
# 声明license路径
os.environ['QLIC'] = '/finlab.basic/kx'

import signal
import time
import traceback
import pykx as kx

RTP_PORT = 5015
TICKERPLANT_PORT = 5010
SUBSCRIPTION_TABLE = 'trade'


# RTP 的回调应该直接注册到 RTP 实例上，
# 而不是只把函数塞进 kx.q 命名空间。
def upd(table, data):
    print("========== UPD ==========")
    print("table:", table)
    print("data:", data)
    print("type:", type(data))

    try:
        df = data.pd()
        print(f"[{table}] 收到 {len(df)} 条新数据:")
        print(df)
    except Exception as e:
        print(f"[{table}] 处理数据失败: {e!r}")
        traceback.print_exc()

    # 在这里将数据发往 Kafka、WebSocket 或策略模型


def preprocessor(table, data):
    # 直接透传所有订阅到的数据
    return data


rtp = kx.tick.RTP(
    port=RTP_PORT,
    subscriptions=[SUBSCRIPTION_TABLE],
    libraries={'kx': 'pykx'},
    pre_processor=preprocessor,
    post_processor=upd,
    vanilla=False,
)
rtp.start({'tickerplant': f'localhost:{TICKERPLANT_PORT}'})

print(f"RTP 订阅已启动，正在监听 {TICKERPLANT_PORT} 端口的数据推送...")


def shutdown(*_args):
    try:
        rtp.stop()
        print(f"RTP 已停止，{RTP_PORT} 端口已释放")
    except Exception as e:
        print(f"停止 RTP 失败: {e!r}")
    raise SystemExit(0)


signal.signal(signal.SIGINT, shutdown)
signal.signal(signal.SIGTERM, shutdown)

while True:
    time.sleep(1)
