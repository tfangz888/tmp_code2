#!/bin/bash
# 检查所有挂载点使用率，超过阈值就记录日志
THRESHOLD=70

df -h | while read filesystem size used avail percent mountpoint; do
    # 跳过标题行和特殊文件系统
    if [[ "$percent" =~ ^[0-9]+%$ ]] && [ "${percent%\%}" -gt $THRESHOLD ]; then
        echo "$(date): 磁盘 $mountpoint 使用率 $percent 超过${THRESHOLD}%" 
    fi
done




sudo nano /etc/systemd/system/disk-check.service

[Unit]
Description=Check Disk Usage
After=multi-user.target

[Service]
Type=oneshot
ExecStart=/usr/local/bin/check-disk.sh

[Install]
WantedBy=multi-user.target
